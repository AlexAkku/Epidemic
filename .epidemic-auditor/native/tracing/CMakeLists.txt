add_library(AuditorExactRuntime STATIC EXCLUDE_FROM_ALL exact_runtime.cpp exact_hooks.asm)
target_compile_features(AuditorExactRuntime PRIVATE cxx_std_17)
target_compile_definitions(AuditorExactRuntime PRIVATE UNICODE _UNICODE)
target_compile_options(AuditorExactRuntime PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/Zi>)
# This directory has no /Gh or /GH: the hook producer must not instrument itself.

function(auditor_collect_targets directory output)
  get_property(targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
  get_property(children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
  foreach(child IN LISTS children)
    auditor_collect_targets("${child}" nested)
    list(APPEND targets ${nested})
  endforeach()
  set(${output} ${targets} PARENT_SCOPE)
endfunction()

function(auditor_apply_coverage directory)
  auditor_collect_targets("${directory}" targets)
  foreach(target IN LISTS targets)
    if(target STREQUAL "AuditorExactRuntime")
      continue()
    endif()
    get_target_property(type "${target}" TYPE)
    if(type MATCHES "^(STATIC|SHARED|MODULE|OBJECT)_LIBRARY$" OR type STREQUAL "EXECUTABLE")
      target_compile_options("${target}" PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/clang:-fprofile-instr-generate;/clang:-fcoverage-mapping>)
      if(type MATCHES "^(SHARED|MODULE)_LIBRARY$" OR type STREQUAL "EXECUTABLE")
        target_link_options("${target}" PRIVATE /INCREMENTAL:NO)
      endif()
    endif()
  endforeach()
endfunction()

function(auditor_apply_exact_tracing directory)
  auditor_collect_targets("${directory}" targets)
  foreach(target IN LISTS targets)
    if(target STREQUAL "AuditorExactRuntime")
      continue()
    endif()
    get_target_property(type "${target}" TYPE)
    if(type MATCHES "^(STATIC|SHARED|MODULE|OBJECT)_LIBRARY$" OR type STREQUAL "EXECUTABLE")
      target_compile_options("${target}" PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/Gh;/GH;/Ob0;/Zi>)
      target_link_libraries("${target}" PRIVATE AuditorExactRuntime)
      if(type MATCHES "^(SHARED|MODULE)_LIBRARY$" OR type STREQUAL "EXECUTABLE")
        target_link_options("${target}" PRIVATE /DEBUG:FULL /INCREMENTAL:NO /OPT:NOICF)
      endif()
    endif()
  endforeach()
endfunction()
