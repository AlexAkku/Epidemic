cmake_minimum_required(VERSION 3.20)
project(EpidemicAuditor LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "EpidemicAuditor native application requires Windows")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
# Compiler products stay in the CMake build tree. Runtime deployment is a copy.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin/$<CONFIG>")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER "${config}" upper_config)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${upper_config} "${CMAKE_CURRENT_BINARY_DIR}/bin/${config}")
endforeach()
get_filename_component(AUDITOR_BUNDLE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.ai/tools/auditor" ABSOLUTE)
get_filename_component(AUDITOR_PYTHON "${CMAKE_CURRENT_SOURCE_DIR}/../../.ai/venv/auditor/Scripts/python.exe" ABSOLUTE)
if(NOT DEFINED Python3_EXECUTABLE AND EXISTS "${AUDITOR_PYTHON}")
  set(Python3_EXECUTABLE "${AUDITOR_PYTHON}")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Embed)
add_executable(EpidemicAuditor WIN32 launcher.cpp)
target_compile_features(EpidemicAuditor PRIVATE cxx_std_17)
target_compile_definitions(EpidemicAuditor PRIVATE UNICODE _UNICODE)
target_link_libraries(EpidemicAuditor PRIVATE Python3::Python shell32 user32)
add_executable(EpidemicAuditorObservation observation.cpp mutex_observation.cpp)
target_compile_features(EpidemicAuditorObservation PRIVATE cxx_std_17)
target_compile_definitions(EpidemicAuditorObservation PRIVATE UNICODE _UNICODE)
target_link_libraries(EpidemicAuditorObservation PRIVATE dbghelp psapi version ole32 advapi32)
add_dependencies(EpidemicAuditor EpidemicAuditorObservation)
add_executable(EpidemicAuditorEtwBroker etw_broker.cpp etw_broker_install.cpp)
target_compile_features(EpidemicAuditorEtwBroker PRIVATE cxx_std_17)
target_compile_definitions(EpidemicAuditorEtwBroker PRIVATE UNICODE _UNICODE)
target_link_libraries(EpidemicAuditorEtwBroker PRIVATE advapi32 shell32 ole32 bcrypt)
add_dependencies(EpidemicAuditor EpidemicAuditorEtwBroker)
add_executable(EpidemicAuditorConcurrencyFixture EXCLUDE_FROM_ALL concurrency_fixture.cpp)
if(MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 8)
  enable_language(ASM_MASM)
  add_subdirectory(tracing)
  add_library(EpidemicAuditorExactDllFixture SHARED EXCLUDE_FROM_ALL tracing/exact_dll_fixture.cpp)
  target_compile_options(EpidemicAuditorExactDllFixture PRIVATE /W4 /WX /Gh /GH /Ob0 /Zi)
  target_link_libraries(EpidemicAuditorExactDllFixture PRIVATE AuditorExactRuntime)
  target_link_options(EpidemicAuditorExactDllFixture PRIVATE /DEBUG:FULL /INCREMENTAL:NO /OPT:NOICF)
  add_executable(EpidemicAuditorExactFixture EXCLUDE_FROM_ALL tracing/exact_fixture.cpp)
  target_compile_features(EpidemicAuditorExactFixture PRIVATE cxx_std_17)
  target_compile_options(EpidemicAuditorExactFixture PRIVATE /W4 /WX /Gh /GH /Ob0 /Zi)
  target_link_libraries(EpidemicAuditorExactFixture PRIVATE AuditorExactRuntime EpidemicAuditorExactDllFixture)
  target_link_options(EpidemicAuditorExactFixture PRIVATE /DEBUG:FULL /INCREMENTAL:NO /OPT:NOICF)
  add_executable(EpidemicAuditorExactAvxFixture EXCLUDE_FROM_ALL tracing/exact_avx_fixture.cpp)
  target_compile_features(EpidemicAuditorExactAvxFixture PRIVATE cxx_std_17)
  target_compile_options(EpidemicAuditorExactAvxFixture PRIVATE /W4 /WX /arch:AVX2 /Gh /GH /Ob0 /Zi)
  target_link_libraries(EpidemicAuditorExactAvxFixture PRIVATE AuditorExactRuntime)
  target_link_options(EpidemicAuditorExactAvxFixture PRIVATE /DEBUG:FULL /INCREMENTAL:NO /OPT:NOICF)
endif()
target_compile_features(EpidemicAuditorConcurrencyFixture PRIVATE cxx_std_20)
target_compile_definitions(EpidemicAuditorConcurrencyFixture PRIVATE UNICODE _UNICODE)
if(MSVC)
  target_compile_options(EpidemicAuditor PRIVATE /W4 /WX)
  target_compile_options(EpidemicAuditorObservation PRIVATE /W4 /WX /Zi)
  # Debug function pointers must refer to code, not incremental-link ILT thunks
  # without source lines; keep helper symbol fixtures deterministic in both modes.
  target_link_options(EpidemicAuditorObservation PRIVATE /DEBUG:FULL /INCREMENTAL:NO)
  target_compile_options(EpidemicAuditorConcurrencyFixture PRIVATE /W4 /WX /Zi)
  target_link_options(EpidemicAuditorConcurrencyFixture PRIVATE /DEBUG:FULL)
  target_compile_options(EpidemicAuditorEtwBroker PRIVATE /W4 /WX /Zi)
  target_link_options(EpidemicAuditorEtwBroker PRIVATE /DEBUG:FULL)
endif()
add_custom_command(TARGET EpidemicAuditor POST_BUILD
  COMMAND "${CMAKE_COMMAND}" -E make_directory "${AUDITOR_BUNDLE_DIR}"
  COMMAND "${CMAKE_COMMAND}" -E copy_if_different
          "$<TARGET_FILE:EpidemicAuditor>"
          "$<TARGET_FILE:EpidemicAuditorObservation>" "$<TARGET_PDB_FILE:EpidemicAuditorObservation>"
          "$<TARGET_FILE:EpidemicAuditorEtwBroker>" "$<TARGET_PDB_FILE:EpidemicAuditorEtwBroker>"
          "${AUDITOR_BUNDLE_DIR}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/bundle_runtime.py"
          --output "${AUDITOR_BUNDLE_DIR}" --executable "${AUDITOR_BUNDLE_DIR}/EpidemicAuditor.exe"
  VERBATIM)
