cmake_minimum_required(VERSION 3.20)
project(EpidemicAuditor LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "EpidemicAuditor native application requires Windows")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Embed)
add_executable(EpidemicAuditor WIN32 launcher.cpp)
target_compile_features(EpidemicAuditor PRIVATE cxx_std_17)
target_compile_definitions(EpidemicAuditor PRIVATE UNICODE _UNICODE)
target_link_libraries(EpidemicAuditor PRIVATE Python3::Python shell32 user32)
add_executable(EpidemicAuditorObservation observation.cpp mutex_observation.cpp)
target_compile_features(EpidemicAuditorObservation PRIVATE cxx_std_17)
target_compile_definitions(EpidemicAuditorObservation PRIVATE UNICODE _UNICODE)
target_link_libraries(EpidemicAuditorObservation PRIVATE dbghelp psapi version ole32 advapi32)
add_dependencies(EpidemicAuditor EpidemicAuditorObservation)
add_executable(EpidemicAuditorEtwBroker etw_broker.cpp etw_broker_install.cpp)
target_compile_features(EpidemicAuditorEtwBroker PRIVATE cxx_std_17)
target_compile_definitions(EpidemicAuditorEtwBroker PRIVATE UNICODE _UNICODE)
target_link_libraries(EpidemicAuditorEtwBroker PRIVATE advapi32 shell32 ole32 bcrypt)
add_dependencies(EpidemicAuditor EpidemicAuditorEtwBroker)
add_executable(EpidemicAuditorConcurrencyFixture EXCLUDE_FROM_ALL concurrency_fixture.cpp)
target_compile_features(EpidemicAuditorConcurrencyFixture PRIVATE cxx_std_20)
target_compile_definitions(EpidemicAuditorConcurrencyFixture PRIVATE UNICODE _UNICODE)
if(MSVC)
  target_compile_options(EpidemicAuditor PRIVATE /W4 /WX)
  target_compile_options(EpidemicAuditorObservation PRIVATE /W4 /WX /Zi)
  target_link_options(EpidemicAuditorObservation PRIVATE /DEBUG:FULL)
  target_compile_options(EpidemicAuditorConcurrencyFixture PRIVATE /W4 /WX /Zi)
  target_link_options(EpidemicAuditorConcurrencyFixture PRIVATE /DEBUG:FULL)
  target_compile_options(EpidemicAuditorEtwBroker PRIVATE /W4 /WX /Zi)
  target_link_options(EpidemicAuditorEtwBroker PRIVATE /DEBUG:FULL)
endif()
add_custom_command(TARGET EpidemicAuditor POST_BUILD
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/bundle_runtime.py"
          --output "$<TARGET_FILE_DIR:EpidemicAuditor>" --executable "$<TARGET_FILE:EpidemicAuditor>"
  VERBATIM)
