cmake_minimum_required(VERSION 3.25)
project(EpidemicEditorHost LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_TESTING OFF CACHE BOOL "Build bridge and production dependencies")
set(EPIDEMIC_BUILD_RUNTIME ON CACHE BOOL "")
set(EPIDEMIC_BUILD_FRAMEWORK ON CACHE BOOL "")
set(EPIDEMIC_BUILD_MANAGED_BRIDGE OFF CACHE BOOL "")
set(EPIDEMIC_WARNINGS_AS_ERRORS ON CACHE BOOL "")
get_filename_component(ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${ENGINE_ROOT}" engine EXCLUDE_FROM_ALL)
add_executable(EpidemicEditorBackend backend.cpp)
target_link_libraries(EpidemicEditorBackend PRIVATE EpidemicEditorUi)
target_include_directories(EpidemicEditorBackend PRIVATE "${ENGINE_ROOT}/Game/Editor/example")
target_compile_definitions(EpidemicEditorBackend PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
epidemic_configure_target(EpidemicEditorBackend)
option(EPIDEMIC_UI_NATIVE_VIEWPORT "Build the optional JVM viewport adapter" OFF)
if(EPIDEMIC_UI_NATIVE_VIEWPORT)
    find_package(JNI REQUIRED COMPONENTS AWT)
    add_library(EpidemicUiViewport SHARED native_viewport.cpp)
    target_include_directories(EpidemicUiViewport PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(EpidemicUiViewport PRIVATE EpidemicRuntimePresentation JNI::AWT psapi)
    target_compile_definitions(EpidemicUiViewport PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
    epidemic_configure_target(EpidemicUiViewport)
endif()
