cmake_minimum_required(VERSION 3.25)
project(EpidemicNativeLauncher LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The sandbox launcher requires Windows/MSVC x64")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
find_package(JNI REQUIRED COMPONENTS JVM)
# Use the pinned publisher's Windows delay-load and sandbox dependency lists.
set(OS_WINDOWS TRUE)
set(CEF_ROOT "${CEF_SDK}")
set(USE_SANDBOX ON CACHE BOOL "" FORCE)
set(USE_ATL OFF CACHE BOOL "" FORCE)
list(APPEND CMAKE_MODULE_PATH "${CEF_SDK}/cmake")
find_package(CEF REQUIRED)
if(NOT EXISTS "${CEF_SDK}/Release/cef_sandbox.lib")
    message(FATAL_ERROR "Verified CEF sandbox SDK is missing")
endif()
add_executable(EpidemicEditor WIN32 launcher.cpp sandbox_status.cpp)
target_include_directories(EpidemicEditor SYSTEM PRIVATE "${CEF_SDK}" ${JNI_INCLUDE_DIRS})
target_compile_definitions(EpidemicEditor PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN PSAPI_VERSION=1)
target_compile_options(EpidemicEditor PRIVATE /W4 /WX /permissive- /utf-8 /guard:cf /EHsc $<$<CONFIG:Release>:/O2>)
target_link_options(EpidemicEditor PRIVATE ${CEF_LINKER_FLAGS} /guard:cf /DYNAMICBASE /HIGHENTROPYVA /NXCOMPAT
    /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/../../editor-host.manifest")
target_link_libraries(EpidemicEditor PRIVATE "${CEF_SDK}/Release/cef_sandbox.lib" ${CEF_SANDBOX_STANDARD_LIBS} ${CEF_STANDARD_LIBS})
